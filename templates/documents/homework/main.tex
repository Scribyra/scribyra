\begin{problem}
  Show that $\sqrt{2}$ is irrational.
\end{problem}
\begin{solution}
  Suppose $\sqrt{2} = p/q$ with $p, q$ integers without a common factor.
  Then $p^2 = 2q^2$, so $p$ is even; write $p = 2k$. Then $q^2 = 2k^2$,
  so $q$ is even as well, a contradiction.
\end{solution}

\begin{problem}
  Let $f(x) = x^3 - 3x$.
  \begin{enumerate}
    \item Find all critical points of $f$.
    \item Classify each as a local maximum or minimum.
  \end{enumerate}
\end{problem}
\begin{solution}
  \begin{enumerate}
    \item $f'(x) = 3x^2 - 3 = 0$ gives $x = \pm 1$.
    \item $f''(x) = 6x$, so $x = -1$ is a local maximum and $x = 1$ a local minimum.
  \end{enumerate}
\end{solution}

\begin{problem}
  Evaluate $\displaystyle\sum_{k=1}^{n} k$.
\end{problem}
\begin{solution}
  Pair the first and last terms: there are $n/2$ pairs, each summing to $n+1$, so
  \[
    \sum_{k=1}^{n} k = \frac{n(n+1)}{2}. \qedhere
  \]
\end{solution}
